import type { ReviewDirection, ReviewQuestion, ReviewSet } from "./hsk1_3-0";

type TranslationRow = [ReviewDirection, string, string, number, string[]?];
type ReorderRow = [string, string[], number, string[]?];

const translations = (prefix: string, rows: TranslationRow[]): ReviewQuestion[] =>
	rows.map(([direction, prompt, answer, lesson, acceptedAnswers], index) => ({
		id: `${prefix}-${index + 1}`,
		direction,
		prompt,
		answer,
		lesson,
		explanation: `Ôn tập từ vựng và ngữ pháp HSK2 Bài ${lesson}.`,
		acceptedAnswers
	}));

const reorders = (prefix: string, rows: ReorderRow[]): ReviewQuestion[] =>
	rows.map(([answer, tokens, lesson, acceptedAnswers], index) => ({
		id: `${prefix}-${index + 1}`,
		direction: "vi2zh",
		type: "reorder",
		prompt: "Sắp xếp các từ thành câu đúng:",
		answer,
		tokens,
		lesson,
		explanation: `Ôn tập trật tự câu HSK2 Bài ${lesson}.`,
		acceptedAnswers
	}));
