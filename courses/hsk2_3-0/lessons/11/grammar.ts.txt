import type { GrammarExercise, GrammarLesson } from "./types";

type Row = {
	id: string;
	zh: string;
	vi: string;
	tokens: string[];
	wrong: [string, string];
	note: string;
};
const exercises = (prefix: string, rows: Row[]): GrammarExercise[] => [
	{
		id: `${prefix}-choice`,
		title: "1. Chọn đáp án đúng",
		type: "choice",
		questions: rows.map((r, i) => {
			const options: [string, string, string] =
				i % 3 === 0
					? [r.zh, ...r.wrong]
					: i % 3 === 1
						? [r.wrong[0], r.zh, r.wrong[1]]
						: [...r.wrong, r.zh];
			return {
				id: `${r.id}c`,
				prompt: `Chọn câu đúng: “${r.vi}”`,
				answer: r.zh,
				options,
				completedSentence: r.zh,
				meaning: r.vi,
				explanation: r.note,
				optionExplanations: Object.fromEntries(
					options.map((o) => [
						o,
						o === r.zh ? `Đúng: ${r.note}` : "Sai: vị trí từ hoặc ý nghĩa ngữ pháp không phù hợp."
					])
				)
			};
		})
	},
	{
		id: `${prefix}-arrange`,
		title: "2. Sắp xếp từ thành câu",
		type: "arrange",
		questions: rows.map((r) => ({
			id: `${r.id}a`,
			prompt: r.vi,
			answer: r.zh,
			tokens: r.tokens,
			completedSentence: r.zh,
			meaning: r.vi,
			explanation: r.note
		}))
	},
	{
		id: `${prefix}-translate`,
		title: "3. Dịch Việt sang Trung",
		type: "input",
		questions: rows.map((r) => ({
			id: `${r.id}t`,
			prompt: r.vi,
			answer: r.zh,
			completedSentence: r.zh,
			meaning: r.vi,
			explanation: r.note
		}))
	}
];
