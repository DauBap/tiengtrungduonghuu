import type { GrammarExercise, GrammarLesson, GrammarQuestion } from "./types";

type ChoiceRow = [string, string, string, [string, string, string], string, string];
type TextRow = [string, string, string, string, string];
type ArrangeRow = [string, string, string, string[], string, string];

const choiceQuestions = (rows: ChoiceRow[]): GrammarQuestion[] =>
	rows.map(([id, prompt, answer, options, meaning, explanation]) => ({
		id,
		prompt,
		answer,
		options,
		completedSentence: answer,
		meaning,
		explanation,
		optionExplanations: Object.fromEntries(
			options.map((option) => [
				option,
				option === answer
					? "Đúng: đáp án phù hợp cả cấu trúc và nghĩa của đề bài."
					: "Sai: cấu trúc hoặc trật tự từ không phù hợp với yêu cầu."
			])
		)
	}));

const inputQuestions = (rows: TextRow[]): GrammarQuestion[] =>
	rows.map(([id, prompt, answer, meaning, explanation]) => ({
		id,
		prompt,
		answer,
		completedSentence: answer,
		meaning,
		explanation
	}));

const arrangeQuestions = (rows: ArrangeRow[]): GrammarQuestion[] =>
	rows.map(([id, prompt, answer, tokens, meaning, explanation]) => ({
		id,
		prompt,
		answer,
		tokens,
		completedSentence: answer,
		meaning,
		explanation
	}));

const exercises = (
	prefix: string,
	choices: ChoiceRow[],
	arranges: ArrangeRow[],
	inputs: TextRow[]
): GrammarExercise[] => [
	{
		id: `${prefix}-choice`,
		title: "1. Chọn đáp án đúng",
		type: "choice",
		questions: choiceQuestions(choices)
	},
	{
		id: `${prefix}-arrange`,
		title: "2. Sắp xếp từ thành câu",
		type: "arrange",
		questions: arrangeQuestions(arranges)
	},
	{
		id: `${prefix}-translate`,
		title: "3. Dịch Việt sang Trung",
		type: "input",
		questions: inputQuestions(inputs)
	}
];
