import type { GrammarExercise, GrammarLesson } from "./types";

type Row = { id: string; zh: string; vi: string; tokens: string[]; wrong: [string, string] };
const exercises = (prefix: string, rows: Row[]): GrammarExercise[] => [
	{
		id: `${prefix}-choice`,
		title: "1. Chọn đáp án đúng",
		type: "choice",
		questions: rows.map((r, i) => ({
			id: `${r.id}c`,
			prompt: `Chọn câu đúng: “${r.vi}”`,
			answer: r.zh,
			options:
				i % 3 === 0
					? [r.zh, ...r.wrong]
					: i % 3 === 1
						? [r.wrong[0], r.zh, r.wrong[1]]
						: [...r.wrong, r.zh],
			completedSentence: r.zh,
			meaning: r.vi,
			explanation: "Đáp án đúng có cấu trúc và trật tự từ phù hợp.",
			optionExplanations: {
				[r.zh]: "Đúng: đúng cấu trúc trọng tâm.",
				[r.wrong[0]]: "Sai: sai trật tự hoặc hình thức cấu trúc.",
				[r.wrong[1]]: "Sai: cách kết hợp từ không đúng."
			}
		}))
	},
	{
		id: `${prefix}-arrange`,
		title: "2. Sắp xếp từ thành câu",
		type: "arrange",
		questions: rows.map((r) => ({
			id: `${r.id}a`,
			prompt: r.vi,
			answer: r.zh,
			tokens: r.tokens,
			completedSentence: r.zh,
			meaning: r.vi,
			explanation: "Sắp xếp theo đúng trật tự câu tiếng Trung."
		}))
	},
	{
		id: `${prefix}-translate`,
		title: "3. Dịch Việt sang Trung",
		type: "input",
		questions: rows.map((r) => ({
			id: `${r.id}t`,
			prompt: r.vi,
			answer: r.zh,
			completedSentence: r.zh,
			meaning: r.vi,
			explanation: "Câu trả lời dùng đúng điểm ngữ pháp của bài."
		}))
	}
];
