import type { GrammarLesson, GrammarQuestion } from "./types";

const choice = (
	id: string,
	prompt: string,
	answer: string,
	options: [string, string, string],
	meaning: string,
	explanation: string
): GrammarQuestion => ({
	id,
	prompt,
	answer,
	options,
	completedSentence: answer,
	meaning,
	explanation,
	optionExplanations: Object.fromEntries(
		options.map((option) => [
			option,
			option === answer
				? "Đúng: đáp án phù hợp cả cấu trúc ngữ pháp lẫn nghĩa của đề bài."
				: "Sai: cấu trúc hoặc trật tự từ không phù hợp với yêu cầu."
		])
	)
});

const input = (
	id: string,
	prompt: string,
	answer: string,
	meaning: string,
	explanation: string
): GrammarQuestion => ({ id, prompt, answer, completedSentence: answer, meaning, explanation });

const arrange = (
	id: string,
	prompt: string,
	answer: string,
	tokens: string[],
	meaning: string,
	explanation: string
): GrammarQuestion => ({
	id,
	prompt,
	answer,
	tokens,
	completedSentence: answer,
	meaning,
	explanation
});
