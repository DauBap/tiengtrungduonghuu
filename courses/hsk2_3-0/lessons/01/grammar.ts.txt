import type { GrammarLesson, GrammarQuestion } from "./types";

const choice = (
	id: string,
	prompt: string,
	answer: string,
	options: [string, string, string],
	meaning: string,
	explanation: string,
	optionExplanations: Record<string, string>
): GrammarQuestion => ({
	id,
	prompt,
	answer,
	options,
	completedSentence: answer,
	meaning,
	explanation,
	optionExplanations
});

const input = (
	id: string,
	prompt: string,
	answer: string,
	meaning: string,
	explanation: string,
	acceptedAnswers?: string[]
): GrammarQuestion => ({
	id,
	prompt,
	answer,
	acceptedAnswers,
	completedSentence: answer,
	meaning,
	explanation
});

const arrange = (
	id: string,
	prompt: string,
	answer: string,
	tokens: string[],
	meaning: string,
	explanation: string
): GrammarQuestion => ({
	id,
	prompt,
	answer,
	tokens,
	completedSentence: answer,
	meaning,
	explanation
});

const commonChoiceReason = (correct: string, wrong1: string, wrong2: string) => ({
	[correct]: "Đúng: cấu trúc và vị trí từ phù hợp với nghĩa của đề bài.",
	[wrong1]: "Sai: trật tự hoặc cấu trúc ngữ pháp không phù hợp.",
	[wrong2]: "Sai: cách diễn đạt này không đúng với nghĩa được yêu cầu."
});
