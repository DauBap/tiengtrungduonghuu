import type { GrammarExercise, GrammarLesson, GrammarQuestion } from "./types";

type C = [string, string, string, [string, string, string], string];
type A = [string, string, string, string[]];
type I = [string, string, string];
const make = (prefix: string, cs: C[], as: A[], ins: I[]): GrammarExercise[] => [
	{
		id: `${prefix}-choice`,
		title: "1. Chọn đáp án đúng",
		type: "choice",
		questions: cs.map(([id, prompt, answer, options, meaning]) => ({
			id,
			prompt,
			answer,
			options,
			meaning,
			completedSentence: answer,
			explanation: "Đáp án đúng giữ đúng vị trí của bổ ngữ và phù hợp với nghĩa đề bài.",
			optionExplanations: Object.fromEntries(
				options.map((o) => [
					o,
					o === answer
						? "Đúng: đúng cấu trúc và ngữ cảnh."
						: "Sai: sai trật tự, hướng hoặc quan hệ ý nghĩa."
				])
			)
		}))
	},
	{
		id: `${prefix}-arrange`,
		title: "2. Sắp xếp từ thành câu",
		type: "arrange",
		questions: as.map(([id, prompt, answer, tokens]) => ({
			id,
			prompt,
			answer,
			tokens,
			meaning: prompt,
			completedSentence: answer,
			explanation: "Sắp xếp theo đúng trật tự của cấu trúc ngữ pháp."
		}))
	},
	{
		id: `${prefix}-translate`,
		title: "3. Dịch Việt sang Trung",
		type: "input",
		questions: ins.map(([id, prompt, answer]) => ({
			id,
			prompt,
			answer,
			meaning: prompt,
			completedSentence: answer,
			explanation: "Câu trả lời dùng đúng cấu trúc trọng tâm của bài."
		}))
	}
];
