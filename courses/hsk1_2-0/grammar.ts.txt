import type { GrammarLesson, GrammarPoint, GrammarQuestion } from "./types";

type Example = [chinese: string, vietnamese: string];
type Practice = {
	choicePrompt: string;
	answer: string;
	wrong: [string, string];
	arrange: [segmentedAnswer: string, vietnamese: string];
	input?: [prompt: string, answer: string, acceptedAnswers?: string[]];
};

const clean = (value: string) => value.replace(/[。.!！?？，,\s]+/g, "");
const question = (
	id: string,
	prompt: string,
	answer: string,
	extra: Partial<GrammarQuestion> = {}
): GrammarQuestion => ({ id, prompt, answer, completedSentence: answer, ...extra });

function point(
	id: string,
	title: string,
	explanation: string,
	formulas: string[],
	examples: Example[],
	practice: Practice,
	notes?: string[]
): GrammarPoint {
	const [segmentedAnswer, arrangeMeaning] = practice.arrange;
	const arrangedAnswer = clean(segmentedAnswer.replaceAll("|", ""));
	let tokens = segmentedAnswer.split("|").filter(Boolean);
	if (tokens.length > 1 && clean(tokens.join("")) === arrangedAnswer) {
		tokens = [...tokens.slice(1), tokens[0]];
	}
	const [inputPrompt, inputAnswer, acceptedAnswers] = practice.input ?? [
		examples[0][1],
		clean(examples[0][0]),
		undefined
	];
	return {
		id,
		title,
		explanation,
		formulas,
		examples: examples.map(([chinese, vietnamese]) => ({ chinese, vietnamese })),
		notes,
		exercises: [
			{
				id: `${id}-choice`,
				title: "1. Chọn đáp án đúng",
				type: "choice",
				questions: [
					question(`${id}-c1`, practice.choicePrompt, practice.answer, {
						options: [practice.answer, ...practice.wrong],
						meaning: examples[0][1],
						explanation,
						formula: formulas[0],
						optionExplanations: {
							[practice.answer]: "Đúng nghĩa, đúng trật tự và đúng điểm ngữ pháp đang luyện.",
							[practice.wrong[0]]: "Phương án này sai cấu trúc hoặc không phù hợp ngữ cảnh.",
							[practice.wrong[1]]:
								"Phương án này làm thay đổi nghĩa hoặc đặt thành phần ngữ pháp sai vị trí."
						}
					})
				]
			},
			{
				id: `${id}-arrange`,
				title: "2. Sắp xếp từ thành câu",
				type: "arrange",
				questions: [
					question(`${id}-a1`, arrangeMeaning, arrangedAnswer, {
						translation: arrangeMeaning,
						tokens,
						meaning: arrangeMeaning,
						explanation: "Ghép các từ theo đúng trật tự của cấu trúc đang học.",
						formula: formulas[0]
					})
				]
			},
			{
				id: `${id}-input`,
				title: "3. Dịch hoặc biến đổi câu",
				type: "input",
				questions: [
					question(`${id}-i1`, inputPrompt, inputAnswer, {
						acceptedAnswers,
						meaning: inputPrompt,
						explanation: "Dấu câu và khoảng trắng không ảnh hưởng đến kết quả chấm.",
						formula: formulas[0]
					})
				]
			}
		]
	};
}
