const choice = (
	id: string,
	prompt: string,
	answer: string,
	options: [string, string, string],
	meaning: string,
	explanation: string,
	optionExplanations: Record<string, string>,
	completedSentence = answer
): GrammarQuestion => ({
	id,
	prompt,
	answer,
	options,
	meaning,
	explanation,
	optionExplanations,
	completedSentence
});
const input = (id: string, prompt: string, answer: string): GrammarQuestion => ({
	id,
	prompt,
	answer,
	completedSentence: answer,
	meaning: prompt,
	explanation: "Câu trả lời giữ đúng trật tự từ và cấu trúc ngữ pháp của đề bài.",
	formula: "Theo công thức của điểm ngữ pháp này."
});
const arrange = (
	id: string,
	prompt: string,
	answer: string,
	tokens: string[]
): GrammarQuestion => ({
	id,
	prompt,
	answer,
	tokens,
	completedSentence: answer,
	meaning: prompt,
	explanation: "Sắp xếp các từ theo đúng trật tự của cấu trúc câu tiếng Trung.",
	formula: "Chủ ngữ + thành phần ngữ pháp + danh từ/cụm danh từ."
});

export function getGrammarLesson(courseId: string, lessonId: number): GrammarLesson | undefined {
	const normalizedCourseId = courseId.toLowerCase();
	if (normalizedCourseId === "hsk1_2") return getHsk12GrammarLesson(lessonId);
	if (["hsk2_3-0", "hsk2_3.0"].includes(normalizedCourseId)) {
		if (lessonId === 1) return hsk2Lesson1Grammar;
		if (lessonId === 2) return hsk2Lesson2Grammar;
		if (lessonId === 3) return hsk2Lesson3Grammar;
		if (lessonId === 4) return hsk2Lesson4Grammar;
		if (lessonId === 5) return hsk2Lesson5Grammar;
		if (lessonId === 6) return hsk2Lesson6Grammar;
		if (lessonId === 7) return hsk2Lesson7Grammar;
		if (lessonId === 8) return hsk2Lesson8Grammar;
		if (lessonId === 9) return hsk2Lesson9Grammar;
		if (lessonId === 10) return hsk2Lesson10Grammar;
		if (lessonId === 11) return hsk2Lesson11Grammar;
		if (lessonId === 12) return hsk2Lesson12Grammar;
		if (lessonId === 13) return hsk2Lesson13Grammar;
		if (lessonId === 14) return hsk2Lesson14Grammar;
		if (lessonId === 15) return hsk2Lesson15Grammar;
		return undefined;
	}
	if (normalizedCourseId !== "hsk1_3-0") return undefined;
	if (lessonId === 3) return hsk1Lesson3Grammar;
	if (lessonId === 4) return hsk1Lesson4Grammar;
	if (lessonId === 5) return hsk1Lesson5Grammar;
	if (lessonId === 6) return hsk1Lesson6Grammar;
	if (lessonId === 7) return hsk1Lesson7Grammar;
	if (lessonId === 8) return hsk1Lesson8Grammar;
	if (lessonId === 9) return hsk1Lesson9Grammar;
	if (lessonId === 10) return hsk1Lesson10Grammar;
	if (lessonId === 11) return hsk1Lesson11Grammar;
	if (lessonId === 12) return hsk1Lesson12Grammar;
	if (lessonId === 13) return hsk1Lesson13Grammar;
	if (lessonId === 14) return hsk1Lesson14Grammar;
	if (lessonId === 15) return hsk1Lesson15Grammar;
	return undefined;
}
