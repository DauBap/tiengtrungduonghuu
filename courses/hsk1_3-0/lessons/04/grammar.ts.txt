import type { GrammarLesson, GrammarQuestion } from "./types";

type Reasons = [string, string, string];

const mc = (
	id: string,
	prompt: string,
	answer: string,
	options: [string, string, string],
	meaning: string,
	explanation: string,
	formula: string,
	reasons: Reasons,
	completedSentence = answer
): GrammarQuestion => ({
	id,
	prompt,
	answer,
	options,
	meaning,
	explanation,
	formula,
	completedSentence,
	optionExplanations: Object.fromEntries(options.map((option, index) => [option, reasons[index]]))
});

const input = (
	id: string,
	prompt: string,
	answer: string,
	meaning: string,
	explanation: string,
	formula: string,
	acceptedAnswers?: string[]
): GrammarQuestion => ({
	id,
	prompt,
	answer,
	acceptedAnswers,
	completedSentence: answer,
	meaning,
	explanation,
	formula
});

	completedSentence: answer,
	meaning,
	explanation,
	formula
});

const numberRecognize = numberConfigs.map(([arabic, hanzi], index) => {
	const wrong1 =
		index % 2 === 0 ? String(Number(arabic) + 10) : String(Math.max(0, Number(arabic) - 1));
	const wrong2 = arabic === "100" ? "1000" : String(Number(arabic) + 1);
	const options = (
		index % 3 === 0
			? [arabic, wrong1, wrong2]
			: index % 3 === 1
				? [wrong1, arabic, wrong2]
				: [wrong1, wrong2, arabic]
	) as [string, string, string];
	return mc(
		`num-c${index + 1}`,
		`“${hanzi}” là số nào?`,
		arabic,
		options,
		`${hanzi}: ${arabic}`,
		"Xác định đúng hàng chục và hàng đơn vị.",
		"Số hàng chục + 十 + số hàng đơn vị",
		options.map((o) => (o === arabic ? "Đúng giá trị." : "Không khớp giá trị chữ Hán.")) as Reasons,
		`${hanzi} = ${arabic}`
	);
});
