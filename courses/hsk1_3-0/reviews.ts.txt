export type ReviewDirection = "vi2zh" | "zh2vi";
export type ReviewQuestionType = "translation" | "reorder";

export type ReviewQuestion = {
	id: string;
	direction: ReviewDirection;
	prompt: string;
	answer: string;
	acceptedAnswers?: string[];
	explanation: string;
	type?: ReviewQuestionType;
	tokens?: string[];
	lesson?: number;
};

export type ReviewSection = {
	id: string;
	title: string;
	questions: ReviewQuestion[];
};

export type ReviewSet = {
	id: string;
	courseId: string;
	title: string;
	subtitle: string;
	questions?: ReviewQuestion[];
	sections?: ReviewSection[];
	legacyQuestions?: ReviewQuestion[];
};

const q = (id: string, direction: ReviewDirection, prompt: string, answer: string, explanation: string, acceptedAnswers: string[] = []): ReviewQuestion => ({
	id,
	direction,
	prompt,
	answer,
	explanation,
	acceptedAnswers: acceptedAnswers.length ? acceptedAnswers : undefined
});

const qA = (id: string, direction: ReviewDirection, prompt: string, answer: string, explanation: string, acceptedAnswers: string[]): ReviewQuestion => ({
	id,
	direction,
	prompt,
	answer,
	explanation,
	acceptedAnswers: acceptedAnswers.length ? acceptedAnswers : undefined
});

const r = (id: string, answer: string, tokens: string[], lesson: number, explanation: string): ReviewQuestion => ({
	id,
	direction: "vi2zh",
	type: "reorder",
	prompt: "Sắp xếp các từ thành câu đúng:",
	answer,
	tokens,
	lesson,
	explanation
});

const ql = (id: string, direction: ReviewDirection, prompt: string, answer: string, lesson: number, explanation: string, acceptedAnswers: string[] = []): ReviewQuestion => ({
	...q(id, direction, prompt, answer, explanation, acceptedAnswers),
	lesson
});

export function getReviewSet(courseId: string, reviewId: string) {
	const normalizedCourse = courseId.toLowerCase();
	const normalizedReview = reviewId.toLowerCase();
	const review = courseReviewTemplates[normalizedCourse]?.[normalizedReview];

	if (!review) {
		return {
			id: String(reviewId),
			courseId: String(courseId),
			title: "ÔN TẬP",
			subtitle: "Đang được cập nhật",
			questions: []
		};
	}

	return review;
}

export function getReviewIds(courseId: string) {
	const normalizedCourse = courseId.toLowerCase();
	return Object.keys(courseReviewTemplates[normalizedCourse] ?? {});
}
